// うちわ受付API v5 // フォーム → ①LINEトークン確認 → ②(画像があれば)Driveに保存 → ③スプレッドシートに記録（画像はリンク＋サムネ表示） // → ④お客さんのLINEへ自動送信（プレビュー画像つき）→ ⑤自分のLINEには「注文」のときだけ通知 // + 進捗確認（受付番号でGET）+ 非常停止スイッチ function doGet(e) { var p = PropertiesService.getScriptProperties(); var no = e.parameter && e.parameter.no; if (!no) return out({ ok: true, msg: 'うちわ受付API 稼働中' }); no = String(no).trim().toUpperCase(); var sh = SpreadsheetApp.openById(p.getProperty('SHEET_ID')).getSheets()[0]; var vals = sh.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (String(vals[i][1]).toUpperCase() === no) { return out({ ok: true, found: true, no: vals[i][1], kind: vals[i][2], name: vals[i][3], total: vals[i][7], qty: vals[i][8], usedate: vals[i][9], status: vals[i][10] }); } } return out({ ok: true, found: false }); } function doPost(e) { var lock = LockService.getScriptLock(); try { var p = PropertiesService.getScriptProperties(); if (p.getProperty('MAINTENANCE') === '1') { return out({ ok: false, error: 'maintenance' }); } var data = JSON.parse(e.postData.contents); var v = JSON.parse(UrlFetchApp.fetch( 'https://api.line.me/oauth2/v2.1/verify?access_token=' + encodeURIComponent(data.token), { muteHttpExceptions: true }).getContentText()); if (String(v.client_id) !== p.getProperty('LOGIN_CHANNEL_ID') || !(v.expires_in > 0)) { return out({ ok: false, error: 'invalid token' }); } var prof = JSON.parse(UrlFetchApp.fetch('https://api.line.me/v2/profile', { headers: { Authorization: 'Bearer ' + data.token }, muteHttpExceptions: true }).getContentText()); if (!prof.userId) return out({ ok: false, error: 'no profile' }); lock.waitLock(20000); var sh = SpreadsheetApp.openById(p.getProperty('SHEET_ID')).getSheets()[0]; if (sh.getLastRow() === 0) { sh.appendRow(['受付日時', '受付番号', '種別', 'お名前', 'LINE名', 'LINEユーザーID', '注文内容', '合計', '枚数', '使用希望日', '進捗', 'デザイン画像', 'デザインサムネ', 'プレビュー画像', 'プレビューサムネ']); } var no = 'UW-' + ('0000' + sh.getLastRow()).slice(-4); var kind = data.kind === 'order' ? '注文' : '見積り'; var imgUrl = saveImage(data.designImage, no + '-design'); var pvUrl = saveImage(data.previewImage, no + '-preview'); sh.appendRow([new Date(), no, kind, safe(data.name), safe(prof.displayName), prof.userId, safe(data.summary), Number(data.total) || 0, Number(data.qty) || 0, safe(data.usedate), '未対応', linkCell(imgUrl), thumbCell(imgUrl), linkCell(pvUrl), thumbCell(pvUrl)]); lock.releaseLock(); var text = (kind === '注文' ? '【ご注文を受け付けました】' : '【お見積り】') + '\n' + '受付番号：' + no + '\n' + String(data.name || '') + ' 様\n\n' + String(data.summary || '') + '\n\n' + '合計：¥' + yen(data.total) + '（' + (Number(data.qty) || 1) + '枚）' + (imgUrl ? '\n\n画像を受け取りました🩵' : ''); var customerMsgs = [{ type: 'text', text: text.slice(0, 4900) }]; if (pvUrl) customerMsgs.push(imageMsg(pvUrl)); var res = pushMessage(p.getProperty('CHANNEL_ACCESS_TOKEN'), prof.userId, customerMsgs); // 自分（お店）への通知：注文が確定したときだけ。見積りのときは何も送らない var ownerId = p.getProperty('OWNER_LINE_ID'); if (kind === '注文' && ownerId) { var ownerText = '【新規注文】' + no + '\n' + String(data.name || '') + ' 様\n合計：¥' + yen(data.total) + '（' + (Number(data.qty) || 1) + '枚）\n使用希望日：' + String(data.usedate || '未入力') + (imgUrl ? '\n持ち込み画像：' + imgUrl : ''); var ownerMsgs = [{ type: 'text', text: ownerText.slice(0, 4900) }]; if (pvUrl) ownerMsgs.push(imageMsg(pvUrl)); if (imgUrl) ownerMsgs.push(imageMsg(imgUrl)); pushMessage(p.getProperty('CHANNEL_ACCESS_TOKEN'), ownerId, ownerMsgs); } return out({ ok: true, receiptNo: no, pushed: res }); } catch (err) { return out({ ok: false, error: String(err) }); } finally { try { lock.releaseLock(); } catch (x) {} } } function saveImage(dataUrl, filename) { if (!dataUrl || String(dataUrl).indexOf('base64,') === -1) return ''; try { var b64 = String(dataUrl).split('base64,')[1]; var blob = Utilities.newBlob(Utilities.base64Decode(b64), 'image/jpeg', filename + '.jpg'); var file = getImageFolder().createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); // サムネ用途にもLINE送信にも使える、Googleの画像専用エンドポイント return 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w1200'; } catch (err) { return ''; } } // スプレッドシート用：クリックできるリンクのセル function linkCell(url) { return url ? '=HYPERLINK("' + url + '","画像を開く")' : ''; } // スプレッドシート用：その場で見えるサムネイルのセル function thumbCell(url) { return url ? '=IFERROR(IMAGE("' + url + '",4,60,60),"")' : ''; } function imageMsg(url) { return { type: 'image', originalContentUrl: url, previewImageUrl: url }; } function pushMessage(token, to, messages) { var r = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', { method: 'post', contentType: 'application/json', headers: { Authorization: 'Bearer ' + token }, payload: JSON.stringify({ to: to, messages: messages }), muteHttpExceptions: true }); return r.getResponseCode() === 200; } function getImageFolder() { var it = DriveApp.getFoldersByName('うちわ受付_画像'); return it.hasNext() ? it.next() : DriveApp.createFolder('うちわ受付_画像'); } function out(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } function safe(s) { s = String(s == null ? '' : s).slice(0, 2000); return /^[=+\-@]/.test(s) ? "'" + s : s; } function yen(n) { return String(Number(n) || 0).replace(/\B(?=(\d{3})+(?!\d))/g, ','); }
